set(BENCH_SIZE_CASES simple many_options subcommand)
set(BENCH_SIZE_LIBS cli20 cli11 argparse cxxopts boost)

foreach(case_name IN LISTS BENCH_SIZE_CASES)
  foreach(lib_name IN LISTS BENCH_SIZE_LIBS)
    set(target_name "bench_size_${case_name}_${lib_name}")
    set(source_file "${case_name}/${lib_name}.cc")
    if(lib_name STREQUAL "cli20")
      set(link_target bench_common)
    elseif(lib_name STREQUAL "cli11")
      set(link_target bench_cli11)
    elseif(lib_name STREQUAL "argparse")
      set(link_target bench_argparse)
    elseif(lib_name STREQUAL "cxxopts")
      set(link_target bench_cxxopts)
    elseif(lib_name STREQUAL "boost")
      set(link_target bench_boost_program_options)
    endif()
    bench_add_size_executable("${target_name}" "${source_file}" "${link_target}")
  endforeach()
endforeach()
