247template <StringLiteral Name>
248[[nodiscard]]
249constexpr auto is_valid_long_option_name() noexcept -> bool {
250 constexpr auto size = Name.size();
251
252 if constexpr (size == 0) {
253 return false;
254 }
255 auto is_alpha = [](char c) constexpr -> bool {
256 return ('a' <= c && c <= 'z');
257 };
258 auto is_digit = [](char c) constexpr -> bool { return '0' <= c && c <= '9'; };
259 auto is_alnum = [&](char c) constexpr -> bool {
260 return is_alpha(c) || is_digit(c);
261 };
262
263 if (!is_alpha(Name[0])) {
264 return false;
265 }
266 bool previous_is_hyphen = false;
267 for (std::size_t i = 1; i < size; ++i) {
268 const char c = Name[i];
269 if (c == '-') {
270 if (previous_is_hyphen) {
271 return false;
272 }
273 previous_is_hyphen = true;
274 continue;
275 }
276 if (!is_alnum(c)) {
277 return false;
278 }
279 previous_is_hyphen = false;
280 }
281
282 return !previous_is_hyphen;
283}
