212 using U = std::remove_cvref_t<T>;
213
214 if constexpr (std::same_as<U, std::string>) {
215 return value;
216 } else if constexpr (std::same_as<U, std::string_view>) {
217 return std::string(value);
218 } else if constexpr (std::same_as<U, const char*>) {
219 return value == nullptr ? std::string{} : std::string(value);
220 } else if constexpr (std::is_same_v<U, bool>) {
221 return value ? "true" : "false";
222 } else if constexpr (std::integral<U>) {
223 return std::to_string(value);

225 return value.string();
226 } else if constexpr (requires { std::format("{}", value); } or
227 std::floating_point<U>) {
228 return std::format("{}", value);
229 } else {
230 return "<value>";
231 }

346struct Action {
347 template <class Arg, class Result>
348 [[nodiscard]]
349 static constexpr auto invoke(ActionCtx<Arg>& ctx, ActionResult<Result> input) {
350 return
351 []<auto FnHead, auto... FnTail>(
352 ActionCtx<Arg>& ctx, ActionResult<Result> input) -> decltype(auto) {
353 using Head = std::remove_cvref_t<decltype(FnHead)>;
354 using Next = typename Head::template after_type<Result>;
355
356 if (!input) {
357 if constexpr (sizeof...(FnTail) == 0) {
358 return ActionResult<Next>::fail(input.error);
359 } else {
360 return Action<FnTail...>::invoke(
361 ctx, ActionResult<Next>::fail(input.error));
362 }
363 }
364
365 auto next = FnHead(ctx, std::move(input));
366 if constexpr (sizeof...(FnTail) == 0) {
367 return next;
368 } else {
369 return Action<FnTail...>::invoke(ctx, std::move(next));
370 }
371 }.template operator()<Fns...>(ctx, std::move(input));
372 }

374 template <class Arg, class Result>
375 [[nodiscard]]
376 static constexpr auto invoke(const ActionCtx<Arg>& ctx,
377 ActionResult<Result> input) {
378 return []<auto FnHead, auto... FnTail>(
379 const ActionCtx<Arg>& ctx,
380 ActionResult<Result> input) -> decltype(auto) {
381 using Head = std::remove_cvref_t<decltype(FnHead)>;
382 using Next = typename Head::template after_type<Result>;
383
384 if (!input) {
385 if constexpr (sizeof...(FnTail) == 0) {
386 return ActionResult<Next>::fail(input.error);
387 } else {
388 return Action<FnTail...>::invoke(
389 ctx, ActionResult<Next>::fail(input.error));
390 }
391 }
392
393 auto next = FnHead(ctx, std::move(input));
394 if constexpr (sizeof...(FnTail) == 0) {
395 return next;
396 } else {
397 return Action<FnTail...>::invoke(ctx, std::move(next));
398 }
399 }.template operator()<Fns...>(ctx, std::move(input));
400 }
401
402 template <class Prev, auto FnHead, auto... FnTail>
403 static constexpr auto validate_impl() {
404 static_assert(
405 sizeof...(FnTail) != 0 ||
406 !std::same_as<typename std::remove_cvref_t<
407 decltype(FnHead)>::template storage_type<Prev>,
408 void>,
409 "The last action must have a non-void storage type");
410
411 using Head = std::remove_cvref_t<decltype(FnHead)>;
412 if constexpr (sizeof...(FnTail) == 0) {
413 return std::make_pair(
414 Head::template accepts_input<Prev>,
415 std::type_identity<typename Head::template storage_type<Prev>>{});
416 } else {
417 if constexpr (!Head::template accepts_input<Prev>) {
418 return std::make_pair(false, std::type_identity<void>{});
419 } else {
420 using Next = typename Head::template after_type<Prev>;
421 return validate_impl<Next, FnTail...>();
422 }
423 }
424 }
425
426 static constexpr auto validate() -> auto {
427 return validate_impl<std::string_view, Fns...>();
428 }
