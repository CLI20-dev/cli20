313 std::floating_point<U>) {

478 template <bool EnableTailCheck, class Prev, auto FnHead, auto... FnTail>
479 static constexpr auto validate_impl() {
480 static_assert(
481 !EnableTailCheck ||
482 (sizeof...(FnTail) != 0 ||
483 !std::same_as<typename std::remove_cvref_t<
484 decltype(FnHead)>::template storage_type<Prev>,
485 void>),
486 "The last action must have a non-void storage type");
487
488 using Head = std::remove_cvref_t<decltype(FnHead)>;
489 if constexpr (sizeof...(FnTail) == 0) {
490 return std::make_tuple(
491 Head::template accepts_input<Prev>,
492 std::type_identity<typename Head::template storage_type<Prev>>{},
493 std::type_identity<typename Head::template after_type<Prev>>{});
494 } else {
495 if constexpr (!Head::template accepts_input<Prev>) {
496 static_assert(Head::template accepts_input<Prev>,
497 "Action step does not accept the previous step's output");
498 return std::make_tuple(false, std::type_identity<void>{},
499 std::type_identity<void>{});
500 } else {
501 using Next = typename Head::template after_type<Prev>;
502 return validate_impl<EnableTailCheck, Next, FnTail...>();
503 }
504 }
