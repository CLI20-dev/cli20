214 if constexpr (std::same_as<U, std::string>) {

347struct Action {
348 template <class Arg, class Result>
349 [[nodiscard]]
350 static constexpr auto invoke(ActionCtx<Arg>& ctx, ActionResult<Result> input) {
351 return
352 []<auto FnHead, auto... FnTail>(
353 ActionCtx<Arg>& ctx, ActionResult<Result> input) -> decltype(auto) {
354 using Head = std::remove_cvref_t<decltype(FnHead)>;
355 using Next = typename Head::template after_type<Result>;
356
357 if (!input) {
358 if constexpr (sizeof...(FnTail) == 0) {
359 return ActionResult<Next>::fail(input.error);
360 } else {
361 return Action<FnTail...>::invoke(
362 ctx, ActionResult<Next>::fail(input.error));
363 }
364 }
365
366 auto next = FnHead(ctx, std::move(input));
367 if constexpr (sizeof...(FnTail) == 0) {
368 return next;
369 } else {
370 return Action<FnTail...>::invoke(ctx, std::move(next));
371 }
372 }.template operator()<Fns...>(ctx, std::move(input));
373 }

376 [[nodiscard]]
377 static constexpr auto invoke(const ActionCtx<Arg>& ctx,
378 ActionResult<Result> input) {

380 const ActionCtx<Arg>& ctx,
381 ActionResult<Result> input) -> decltype(auto) {
382 using Head = std::remove_cvref_t<decltype(FnHead)>;
383 using Next = typename Head::template after_type<Result>;
384
385 if (!input) {
386 if constexpr (sizeof...(FnTail) == 0) {
387 return ActionResult<Next>::fail(input.error);
388 } else {
389 return Action<FnTail...>::invoke(
390 ctx, ActionResult<Next>::fail(input.error));
391 }
392 }
393
394 auto next = FnHead(ctx, std::move(input));
395 if constexpr (sizeof...(FnTail) == 0) {
396 return next;
397 } else {
398 return Action<FnTail...>::invoke(ctx, std::move(next));
399 }
400 }.template operator()<Fns...>(ctx, std::move(input));
401 }

403 template <class Prev, auto FnHead, auto... FnTail>
404 static constexpr auto validate_impl() {
405 static_assert(
406 sizeof...(FnTail) != 0 ||
407 !std::same_as<typename std::remove_cvref_t<
408 decltype(FnHead)>::template storage_type<Prev>,
409 void>,
410 "The last action must have a non-void storage type");
411
412 using Head = std::remove_cvref_t<decltype(FnHead)>;
413 if constexpr (sizeof...(FnTail) == 0) {
414 return std::make_pair(
415 Head::template accepts_input<Prev>,
416 std::type_identity<typename Head::template storage_type<Prev>>{});
417 } else {
418 if constexpr (!Head::template accepts_input<Prev>) {
419 return std::make_pair(false, std::type_identity<void>{});
420 } else {
421 using Next = typename Head::template after_type<Prev>;
422 return validate_impl<Next, FnTail...>();
423 }
424 }
425 }
426
427 static constexpr auto validate() -> auto {
428 return validate_impl<std::string_view, Fns...>();
429 }
