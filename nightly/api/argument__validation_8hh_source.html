172template <class T, std::size_t I = 0>
173consteval auto relation_membership_count(std::string_view name) -> std::size_t {
174 constexpr auto size =
175 std::tuple_size_v<std::remove_cvref_t<decltype(T::relations.items)>>;
176 if constexpr (I >= size) {
177 return 0;
178 } else {
179 constexpr auto rel = std::get<I>(T::relations.items);
180 using R = std::remove_cvref_t<decltype(rel)>;
181 std::size_t current = 0;
182 if constexpr (std::same_as<R, GroupRelation>) {
183 for (auto member : rel.members) {
184 if (member == name) ++current;
185 }
186 } else if constexpr (std::same_as<R, ConflictsRelation>) {
187 if (rel.left == name) ++current;
188 if (rel.right == name) ++current;
189 } else if constexpr (std::same_as<R, DependsOnRelation>) {
190 if (rel.source == name) ++current;
191 if (rel.target == name) ++current;
192 }
193 return current + relation_membership_count<T, I + 1>(name);
194 }
195}

208template <class T, std::size_t I = 0>
209consteval auto relation_items_are_well_formed() -> bool {
210 constexpr auto size =
211 std::tuple_size_v<std::remove_cvref_t<decltype(T::relations.items)>>;
212 if constexpr (I >= size) {
213 return true;
214 } else {
215 constexpr auto rel = std::get<I>(T::relations.items);
216 using R = std::remove_cvref_t<decltype(rel)>;
217 constexpr auto field_names = named_relation_targets<T>();
218 if constexpr (std::same_as<R, GroupRelation>) {
219 if (rel.name.empty() || rel.members.size < 2 ||
220 contains_name(field_names, rel.name) ||
221 group_name_exists_before<T, I>(rel.name)) {
222 return false;
223 }
224 for (auto member : rel.members) {
225 if (!contains_name(field_names, member)) return false;
226 }
227 } else if constexpr (std::same_as<R, ConflictsRelation>) {
228 if (rel.left.empty() || rel.right.empty() || rel.left == rel.right ||
229 !operand_exists<T>(rel.left) || !operand_exists<T>(rel.right)) {
230 return false;
231 }
232 } else if constexpr (std::same_as<R, DependsOnRelation>) {
233 if (rel.source.empty() || rel.target.empty() || rel.source == rel.target ||
234 !operand_exists<T>(rel.source) || !operand_exists<T>(rel.target)) {
235 return false;
236 }
237 } else {
238 return false;
239 }
240 return relation_items_are_well_formed<T, I + 1>();
241 }

254template <StringLiteral Name>
255[[nodiscard]]
256constexpr auto is_valid_long_option_name() noexcept -> bool {
257 constexpr auto size = Name.size();
258
259 if constexpr (size == 0) {
260 return false;
261 }
262 auto is_alpha = [](char c) constexpr -> bool {
263 return ('a' <= c && c <= 'z');
264 };
265 auto is_digit = [](char c) constexpr -> bool { return '0' <= c && c <= '9'; };
266 auto is_alnum = [&](char c) constexpr -> bool {
267 return is_alpha(c) || is_digit(c);
268 };
269
270 if (!is_alpha(Name[0])) {
271 return false;
272 }
273 bool previous_is_hyphen = false;
274 for (std::size_t i = 1; i < size; ++i) {
275 const char c = Name[i];
276 if (c == '-') {
277 if (previous_is_hyphen) {
278 return false;
279 }
280 previous_is_hyphen = true;
281 continue;
282 }
283 if (!is_alnum(c)) {
284 return false;
285 }
286 previous_is_hyphen = false;
287 }
288
289 return !previous_is_hyphen;
290}
