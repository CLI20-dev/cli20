176inline auto tokenize(std::span<const std::string_view> args,
177 const std::unordered_map<std::string, Nargs>& spec_map,
178 const std::unordered_set<std::string>& command_names = {},
179 const TokenizerConfig& cfg = {}) -> TokenizeResult {
180 TokenizeResult result;
181
182 // Returns the matched prefix (as a view into cfg.option_prefixes) or empty.
183 const auto find_prefix = [&](std::string_view tok) -> std::string_view {
184 for (const auto& p : cfg.option_prefixes) {
185 if (tok.size() > p.size() && tok.starts_with(p)) return p;
186 }
187 if (tok.size() > cfg.short_option_prefix.size() &&
188 tok.starts_with(cfg.short_option_prefix)) {
189 return cfg.short_option_prefix;
190 }
191 return {};
192 };
193
194 const auto push = [&](TokenType type, std::string_view text, std::size_t pos,
195 std::string_view prefix = {}) -> void {
196 result.tokens.push_back(
197 {.type = type, .text = text, .matched_prefix = prefix, .position = pos});
198 };
199
200 bool post_separator = false;
201 std::size_t i = 0;
202
203 while (i < args.size()) {
204 const std::string_view tok = args[i];
205
206 // ── option parsing terminator ─────────────────────────────────────────
207 if (!post_separator && tok == cfg.end_of_options_separator) {
208 post_separator = true;
209 ++i;
210 continue;
211 }
212
213 // ── after separator: everything is positional ─────────────────────────
214 if (post_separator) {
215 push(TokenType::positional, tok, i++);
216 continue;
217 }
218
219 // ── command name ──────────────────────────────────────────────────────
220 const auto prefix = find_prefix(tok);
221 if (prefix.empty() && command_names.contains(std::string(tok))) {
222 push(TokenType::command, tok, i);
223 break; // args[i+1:] are the sub-command's responsibility
224 }
225
226 // ── option-looking token ──────────────────────────────────────────────
227 if (!prefix.empty()) {
228 const auto sep = cfg.inline_value_separator;
229 const auto eq_pos = tok.find(sep);
230 const bool has_inline_val = (eq_pos != std::string_view::npos);
231 const std::string_view opt_sv =
232 has_inline_val ? tok.substr(0, eq_pos) : tok;
233 const std::string opt_name{opt_sv};
234
235 const auto spec_it = spec_map.find(opt_name);
236 if (spec_it == spec_map.end()) {
237 return result.fail(ErrorCode::unknown_option, i, opt_name);
238 }
239
240 const Nargs& nargs = spec_it->second;
241 const bool is_flag = (nargs.min == 0 && nargs.max == 0);
242
243 // Use opt_sv (view into args) for matched_prefix so it stays valid
244 // after cfg is destroyed — do not point into cfg.option_prefixes.
245 push(TokenType::option, opt_sv, i, opt_sv.substr(0, prefix.size()));
246
247 if (has_inline_val) {
248 if (is_flag) {
249 return result.fail(ErrorCode::invalid_value, i, opt_name,
250 "flag does not accept a value");
251 }
252 if (nargs.min > 1) {
253 return result.fail(
254 ErrorCode::missing_value, i, opt_name,
255 std::format("option requires at least {} value(s), but inline "
256 "separator provides only 1",
257 nargs.min));
258 }
259 push(TokenType::value, tok.substr(eq_pos + 1), i);
260 ++i;
261 continue;
262 }
263
264 // prefix-opt form: advance, then greedily consume value tokens
265 ++i;
266
267 if (is_flag) continue;
268
269 int count = 0;
270 while (i < args.size()) {
271 const std::string_view next = args[i];
272
273 if (next == cfg.end_of_options_separator) break;
274 if (command_names.contains(std::string(next))) break;
275 if (nargs.max != -1 && count >= nargs.max) break;
276
277 if (!find_prefix(next).empty()) {
278 const auto next_sep = next.find(sep);
279 const std::string next_opt{next_sep != std::string_view::npos
280 ? next.substr(0, next_sep)
281 : next};
282 if (spec_map.contains(next_opt)) break;
283
284 // Unknown option-looking: consume only as first value when min > 0
285 if (count >= 1 || nargs.min == 0) break;
286 }
287
288 push(TokenType::value, args[i], i);
289 ++i;
290 ++count;
291 }
292
293 if (count < nargs.min) {
294 return result.fail(
295 ErrorCode::missing_value, i, opt_name,
296 std::format("option requires at least {} value(s), but {} provided",
297 nargs.min, count));
298 }
299 continue;
300 }
301
302 // ── bare token → positional ───────────────────────────────────────────
303 push(TokenType::positional, tok, i++);
304 }
305
306 return result;
307}

567 auto parse_body(ParseResult<T>& result, std::span<const std::string_view> args,
568 std::size_t first_index) -> void {
569 auto [spec_map, command_names] = get_option_spec_map(result.value);
570 auto tokenized = tokenize(args.subspan(std::min(first_index, args.size())),
571 spec_map, command_names, cfg_);
572 if (tokenized.has_error()) {
573 if (tokenized.error.has_position()) {
574 tokenized.error.position += static_cast<int>(first_index);
575 }
576 result.error = std::move(tokenized.error);
577 return;
578 }
579
580 const auto& tokens = tokenized.tokens;
581 std::size_t pos_idx = 0, pos_cnt = 0;
582
583 for (std::size_t i = 0; i < tokens.size();) {
584 const Token& tok = tokens[i];
585
586 if (tok.type == TokenType::option) {
587 std::size_t j = i + 1;
588 while (j < tokens.size() && tokens[j].type == TokenType::value) ++j;
589 auto err = dispatch_option(result.value, tok.matched_prefix,
590 tok.text.substr(tok.matched_prefix.size()),
591 std::span(tokens).subspan(i + 1, j - i - 1),
592 first_index, tok.position);
593 if (err.has_error()) {
594 result.error = finalize_special_error(std::move(err.error));
595 return;
596 }
597 i = j;
598
599 } else if (tok.type == TokenType::positional) {
600 auto err = dispatch_positional(result.value, pos_idx, pos_cnt, tok.text,
601 tok.position + first_index);
602 if (err.has_error()) {
603 result.error = finalize_special_error(std::move(err.error));
604 return;
605 }
606 ++i;
607
608 } else if (tok.type == TokenType::command) {
609 auto err = dispatch_command(result.value, tok.text, args,
610 tok.position + first_index + 1);
611 if (err.has_error()) {
612 result.error = finalize_special_error(std::move(err.error));
613 return;
614 }
615 break;
616
617 } else {
618 ++i;
619 }
620 }
621
622 // Fire on_parse callbacks for positionals once, after all tokens are
623 // consumed.
624 for_each_field(result.value, [](auto& f) -> auto {
625 using F = std::remove_cvref_t<decltype(f)>;
626 if constexpr (std::derived_from<F, PositionalTag>) {
627 if (f.provided()) f.fire_on_parse();
628 }
629 });
630
631 if (auto err = validate_required(result.value); err.has_error()) {
632 result.error = std::move(err.error);
633 }
634 }

680 // Build the option spec-map and command-name set from the field types of val.
681 auto get_option_spec_map(const T& val)
682 -> std::pair<std::unordered_map<std::string, Nargs>,
683 std::unordered_set<std::string>> {
684 std::unordered_map<std::string, Nargs> spec_map;
685 std::unordered_set<std::string> command_names;
686 for_each_field(val, [&](const auto& f) -> auto {
687 using F = std::remove_cvref_t<decltype(f)>;
688 if constexpr (std::derived_from<F, OptionTag>) {
689 for (const auto& p : cfg_.option_prefixes)
690 spec_map.emplace(p + std::string(F::name.view()), F::nargs);
691 if constexpr (F::short_name != '\0')
692 spec_map.emplace(
693 cfg_.short_option_prefix + std::string(1, F::short_name),
694 F::nargs);
695 }
696 if constexpr (std::derived_from<F, CommandTag>)
697 command_names.emplace(std::string(F::command_name()));
698 });
699 return {spec_map, command_names};
700 }
701
702 // Find the option field matching (prefix, bare), notify it, and invoke its
703 // action on each value token. notify_option_seen() is called once;
704 // invoke_action() (or invoke_flag() for nargs={0,0}) once per value token.
705 auto dispatch_option(T& val, std::string_view prefix, std::string_view bare,
706 std::span<const Token> vals, std::size_t first_index,
707 std::size_t option_position) -> ActionResult<void> {
708 ActionResult<void> res = ActionResult<void>::ok();
709 bool found = false;
710 for_each_field(val, [&](auto& f) -> auto {
711 using F = std::remove_cvref_t<decltype(f)>;
712 if constexpr (std::derived_from<F, OptionTag>) {
713 if (found) return;
714 bool match = false;
715 for (const auto& p : cfg_.option_prefixes)
716 if (p == prefix && F::name.view() == bare) {
717 match = true;
718 break;
719 }
720 if (!match && F::short_name != '\0' &&
721 cfg_.short_option_prefix == prefix && bare.size() == 1 &&
722 bare[0] == F::short_name)
723 match = true;
724 if (!match) return;
725 found = true;
726 f.notify_option_seen();
727 if constexpr (F::nargs.min == 0 && F::nargs.max == 0) {
728 res = f.invoke_flag(option_position + first_index);
729 } else {
730 for (const auto& vt : vals) {
731 if (res.has_error()) break;
732 res = f.invoke_action(vt.text, vt.position + first_index);
733 }
734 }
735 if (!res.has_error()) f.fire_on_parse();
736 }
737 });
738 return res;
739 }

741 // Fill the pos_idx-th positional field with text. Advances pos_idx/pos_cnt
742 // when the current field reaches its nargs.max.
743 static auto dispatch_positional(T& val, std::size_t& pos_idx,
744 std::size_t& pos_cnt, std::string_view text,
745 std::size_t position) -> ActionResult<void> {
746 ActionResult<void> res = ActionResult<void>::ok();
747 bool invoked = false;
748 std::size_t cur = 0;
749 for_each_field(val, [&](auto& f) -> auto {
750 using F = std::remove_cvref_t<decltype(f)>;
751 if constexpr (std::derived_from<F, PositionalTag>) {
752 if (cur++ != pos_idx || invoked) return;
753 invoked = true;
754 res = f.invoke_action(text, position);
755 if (res.has_value()) {
756 ++pos_cnt;
757 if (F::nargs.max != -1 &&
758 pos_cnt >= static_cast<std::size_t>(F::nargs.max)) {
759 ++pos_idx;
760 pos_cnt = 0;
761 }
762 }
763 }
764 });
765 if (!invoked)
766 return ActionResult<void>::fail(
767 ParseError{.code = ErrorCode::unexpected_argument,
768 .kind = ErrorKind::parse,
769 .position = static_cast<int>(position),
770 .subject = std::string(text)});
771 return res;
772 }

774 auto dispatch_command(T& val, std::string_view name,
775 std::span<const std::string_view> args,
776 std::size_t first_index) -> ActionResult<void> {
777 ActionResult<void> res = ActionResult<void>::ok();
778 bool found = false;
779 for_each_field(val, [&](auto& f) -> auto {
780 using F = std::remove_cvref_t<decltype(f)>;
781 if constexpr (std::derived_from<F, CommandTag>) {
782 if (found || F::command_name() != name) {
783 return;
784 }
785 found = true;
786 f.mark_provided();
787 auto sub_parser = Parser<typename F::argument_type>{};
788 sub_parser.program_name_ =
789 std::format("{} {}", program_name_, F::command_name());
790 auto sub = sub_parser.parse(args, first_index);
791 if (sub.has_error()) {
792 res = ActionResult<void>::fail(std::move(sub.error));
793 } else {
794 static_cast<typename F::argument_type&>(f) = std::move(sub.value);
795 f.mark_provided();
796 }
797 }
798 });
799 if (!found) {
800 return ActionResult<void>::fail(ParseError{
801 .code = ErrorCode::unknown_command,
802 .kind = ErrorKind::parse,
803 .position = static_cast<int>(first_index - 1),
804 .subject = std::string(name),
805 });
806 }
807 return res;
808 }

810 static auto validate_required(T& val) -> ActionResult<void> {
811 ActionResult<void> res = ActionResult<void>::ok();
812 for_each_field(val, [&](auto& f) -> auto {
813 using F = std::remove_cvref_t<decltype(f)>;
814 if constexpr (std::derived_from<F, OptionTag> ||
815 std::derived_from<F, PositionalTag>) {
816 if (res.has_error()) {
817 return;
818 }
819 if (f.presence == Presence::required && !f.provided()) {
820 std::string subject;
821 if constexpr (std::derived_from<F, OptionTag>) {
822 subject = std::string(F::name.view());
823 } else {
824 subject = "<positional>";
825 }
826 res = ActionResult<void>::fail(ParseError{
827 .code = ErrorCode::missing_required,
828 .kind = ErrorKind::parse,
829 .subject = std::move(subject),
830 });
831 }
832 }
833 });
834 return res;
835 }
