179inline auto tokenize(std::span<const std::string_view> args,
180 const std::unordered_map<std::string, Nargs>& spec_map,
181 const std::unordered_set<std::string>& command_names = {},
182 const TokenizerConfig& cfg = {}) -> TokenizeResult {
183 TokenizeResult result;
184
185 // Returns the matched prefix (as a view into cfg.option_prefixes) or empty.
186 const auto find_prefix = [&](std::string_view tok) -> std::string_view {
187 for (const auto& p : cfg.option_prefixes) {
188 if (tok.size() > p.size() && tok.starts_with(p)) return p;
189 }
190 if (tok.size() > cfg.short_option_prefix.size() &&
191 tok.starts_with(cfg.short_option_prefix)) {
192 return cfg.short_option_prefix;
193 }
194 return {};
195 };
196
197 const auto push = [&](TokenType type, std::string_view text, std::size_t pos,
198 std::string_view prefix = {}) -> void {
199 result.tokens.push_back(
200 {.type = type, .text = text, .matched_prefix = prefix, .position = pos});
201 };
202
203 bool post_separator = false;
204 std::size_t i = 0;
205
206 while (i < args.size()) {
207 const std::string_view tok = args[i];
208
209 // ── option parsing terminator ─────────────────────────────────────────
210 if (!post_separator && tok == cfg.end_of_options_separator) {
211 post_separator = true;
212 ++i;
213 continue;
214 }
215
216 // ── after separator: everything is positional ─────────────────────────
217 if (post_separator) {
218 push(TokenType::positional, tok, i++);
219 continue;
220 }
221
222 // ── command name ──────────────────────────────────────────────────────
223 const auto prefix = find_prefix(tok);
224 if (prefix.empty() && command_names.contains(std::string(tok))) {
225 push(TokenType::command, tok, i);
226 break; // args[i+1:] are the sub-command's responsibility
227 }
228
229 // ── option-looking token ──────────────────────────────────────────────
230 if (!prefix.empty()) {
231 const auto sep = cfg.inline_value_separator;
232 const auto eq_pos = tok.find(sep);
233 const bool has_inline_val = (eq_pos != std::string_view::npos);
234 const std::string_view opt_sv =
235 has_inline_val ? tok.substr(0, eq_pos) : tok;
236 const std::string opt_name{opt_sv};
237
238 const auto spec_it = spec_map.find(opt_name);
239 if (spec_it == spec_map.end()) {
240 // Short cluster (-xvf) or attached value (-ofile): only when the token
241 // uses the short prefix and carries more than one character after it.
242 const std::string_view body = opt_sv.substr(prefix.size());
243 if (prefix == cfg.short_option_prefix && !has_inline_val &&
244 body.size() > 1) {
245 bool advanced_i = false;
246 const std::string_view char_prefix = tok.substr(0, prefix.size());
247 std::string char_full{char_prefix};
248 char_full.push_back('\0');
249 for (std::size_t k = 0; k < body.size(); ++k) {
250 const std::string_view char_bare = tok.substr(prefix.size() + k, 1);
251 char_full.back() = body[k];
252 const auto it = spec_map.find(char_full);
253 if (it == spec_map.end()) {
254 return result.fail(ErrorCode::unknown_option, i, char_full);
255 }
256 const Nargs& nargs = it->second;
257 const bool is_flag = (nargs.min == 0 && nargs.max == 0);
258 push(TokenType::option, char_bare, i, char_prefix);
259 if (!is_flag) {
260 if (k + 1 < body.size()) {
261 // First value is attached (rest of body).
262 push(TokenType::value, tok.substr(prefix.size() + k + 1), i);
263 ++i;
264 // Consume additional tokens for nargs > 1.
265 int count = 1;
266 while (i < args.size()) {
267 const std::string_view next = args[i];
268 if (next == cfg.end_of_options_separator) break;
269 if (command_names.contains(std::string(next))) break;
270 if (nargs.max != -1 && count >= nargs.max) break;
271 if (!find_prefix(next).empty()) {
272 const auto next_sep = next.find(sep);
273 const std::string next_opt{next_sep != std::string_view::npos
274 ? next.substr(0, next_sep)
275 : next};
276 if (spec_map.contains(next_opt)) break;
277 if (count >= 1 || nargs.min == 0) break;
278 }
279 push(TokenType::value, args[i], i);
280 ++i;
281 ++count;
282 }
283 if (count < nargs.min) {
284 return result.fail(
285 ErrorCode::missing_value, i, char_full,
286 std::format("option requires at least {} value(s), but {} "
287 "provided",
288 nargs.min, count));
289 }
290 } else {
291 // Consume next token(s) greedily as values
292 ++i;
293 int count = 0;
294 while (i < args.size()) {
295 const std::string_view next = args[i];
296 if (next == cfg.end_of_options_separator) break;
297 if (command_names.contains(std::string(next))) break;
298 if (nargs.max != -1 && count >= nargs.max) break;
299 if (!find_prefix(next).empty()) {
300 const auto next_sep = next.find(sep);
301 const std::string next_opt{next_sep != std::string_view::npos
302 ? next.substr(0, next_sep)
303 : next};
304 if (spec_map.contains(next_opt)) break;
305 if (count >= 1 || nargs.min == 0) break;
306 }
307 push(TokenType::value, args[i], i);
308 ++i;
309 ++count;
310 }
311 if (count < nargs.min) {
312 return result.fail(
313 ErrorCode::missing_value, i, char_full,
314 std::format("option requires at least {} value(s), but {} "
315 "provided",
316 nargs.min, count));
317 }
318 }
319 advanced_i = true;
320 break;
321 }
322 }
323 if (!advanced_i) ++i;
324 continue;
325 }
326 return result.fail(ErrorCode::unknown_option, i, opt_name);
327 }
328
329 const Nargs& nargs = spec_it->second;
330 const bool is_flag = (nargs.min == 0 && nargs.max == 0);
331
332 // Store bare name (without prefix) in text; matched_prefix holds the
333 // prefix.
334 push(TokenType::option, opt_sv.substr(prefix.size()), i,
335 opt_sv.substr(0, prefix.size()));
336
337 if (has_inline_val) {
338 if (is_flag) {
339 return result.fail(ErrorCode::invalid_value, i, opt_name,
340 "flag does not accept a value");
341 }
342 if (nargs.min > 1) {
343 return result.fail(
344 ErrorCode::missing_value, i, opt_name,
345 std::format("option requires at least {} value(s), but inline "
346 "separator provides only 1",
347 nargs.min));
348 }
349 push(TokenType::value, tok.substr(eq_pos + 1), i);
350 ++i;
351 continue;
352 }
353
354 // prefix-opt form: advance, then greedily consume value tokens
355 ++i;
356
357 if (is_flag) continue;
358
359 int count = 0;
360 while (i < args.size()) {
361 const std::string_view next = args[i];
362
363 if (next == cfg.end_of_options_separator) break;
364 if (command_names.contains(std::string(next))) break;
365 if (nargs.max != -1 && count >= nargs.max) break;
366
367 if (!find_prefix(next).empty()) {
368 const auto next_sep = next.find(sep);
369 const std::string next_opt{next_sep != std::string_view::npos
370 ? next.substr(0, next_sep)
371 : next};
372 if (spec_map.contains(next_opt)) break;
373
374 // Unknown option-looking: consume only as first value when min > 0
375 if (count >= 1 || nargs.min == 0) break;
376 }
377
378 push(TokenType::value, args[i], i);
379 ++i;
380 ++count;
381 }
382
383 if (count < nargs.min) {
384 return result.fail(
385 ErrorCode::missing_value, i, opt_name,
386 std::format("option requires at least {} value(s), but {} provided",
387 nargs.min, count));
388 }
389 continue;
390 }
391
392 // ── bare token → positional ───────────────────────────────────────────
393 push(TokenType::positional, tok, i++);
394 }
395
396 return result;
397}

657 auto parse_body(ParseResult<T>& result, std::span<const std::string_view> args,
658 std::size_t first_index) -> void {
659 auto [spec_map, command_names] = get_option_spec_map(result.value);
660 auto tokenized = tokenize(args.subspan(std::min(first_index, args.size())),
661 spec_map, command_names, cfg_);
662 if (tokenized.has_error()) {
663 if (tokenized.error.has_position()) {
664 tokenized.error.position += static_cast<int>(first_index);
665 }
666 result.error = std::move(tokenized.error);
667 return;
668 }
669
670 const auto& tokens = tokenized.tokens;
671 std::size_t pos_idx = 0, pos_cnt = 0;
672
673 for (std::size_t i = 0; i < tokens.size();) {
674 const Token& tok = tokens[i];
675
676 if (tok.type == TokenType::option) {
677 std::size_t j = i + 1;
678 while (j < tokens.size() && tokens[j].type == TokenType::value) ++j;
679 auto err = dispatch_option(result.value, tok.matched_prefix, tok.text,
680 std::span(tokens).subspan(i + 1, j - i - 1),
681 first_index, tok.position);
682 if (err.has_error()) {
683 result.error = finalize_special_error(std::move(err.error));
684 return;
685 }
686 i = j;
687
688 } else if (tok.type == TokenType::positional) {
689 auto err = dispatch_positional(result.value, pos_idx, pos_cnt, tok.text,
690 tok.position + first_index);
691 if (err.has_error()) {
692 result.error = finalize_special_error(std::move(err.error));
693 return;
694 }
695 ++i;
696
697 } else if (tok.type == TokenType::command) {
698 auto err = dispatch_command(result.value, tok.text, args,
699 tok.position + first_index + 1);
700 if (err.has_error()) {
701 result.error = finalize_special_error(std::move(err.error));
702 return;
703 }
704 break;
705
706 } else {
707 ++i;
708 }
709 }
710
711 // Fire on_parse callbacks for positionals once, after all tokens are
712 // consumed.
713 for_each_field(result.value, [](auto& f) -> auto {
714 using F = std::remove_cvref_t<decltype(f)>;
715 if constexpr (std::derived_from<F, PositionalTag>) {
716 if (f.provided()) f.fire_on_parse();
717 }
718 });
719
720 if (auto err = apply_env_fallback(result.value); err.has_error()) {
721 result.error = std::move(err.error);
722 return;
723 }
724
725 if (auto err = validate_required(result.value); err.has_error()) {
726 result.error = std::move(err.error);
727 }

774 // Build the option spec-map and command-name set from the field types of val.
775 auto get_option_spec_map(const T& val)
776 -> std::pair<std::unordered_map<std::string, Nargs>,
777 std::unordered_set<std::string>> {
778 std::unordered_map<std::string, Nargs> spec_map;
779 std::unordered_set<std::string> command_names;
780 for_each_field(val, [&](const auto& f) -> auto {
781 using F = std::remove_cvref_t<decltype(f)>;
782 if constexpr (std::derived_from<F, OptionTag>) {
783 for (const auto& p : cfg_.option_prefixes)
784 spec_map.emplace(p + std::string(F::name.view()), F::nargs);
785 if constexpr (F::short_name != '\0')
786 spec_map.emplace(
787 cfg_.short_option_prefix + std::string(1, F::short_name),
788 F::nargs);
789 }
790 if constexpr (std::derived_from<F, CommandTag>)
791 command_names.emplace(std::string(F::command_name()));
792 });
793 return {spec_map, command_names};
794 }

796 // Find the option field matching (prefix, bare), notify it, and invoke its
797 // action on each value token. notify_option_seen() is called once;
798 // invoke_action() (or invoke_flag() for nargs={0,0}) once per value token.
799 auto dispatch_option(T& val, std::string_view prefix, std::string_view bare,
800 std::span<const Token> vals, std::size_t first_index,
801 std::size_t option_position) -> ActionResult<void> {
802 ActionResult<void> res = ActionResult<void>::ok();
803 bool found = false;
804 for_each_field(val, [&](auto& f) -> auto {
805 using F = std::remove_cvref_t<decltype(f)>;
806 if constexpr (std::derived_from<F, OptionTag>) {
807 if (found) return;
808 bool match = false;
809 for (const auto& p : cfg_.option_prefixes)
810 if (p == prefix && F::name.view() == bare) {
811 match = true;
812 break;
813 }
814 if (!match && F::short_name != '\0' &&
815 cfg_.short_option_prefix == prefix && bare.size() == 1 &&
816 bare[0] == F::short_name)
817 match = true;
818 if (!match) return;
819 found = true;
820 f.notify_option_seen();
821 if constexpr (F::nargs.min == 0 && F::nargs.max == 0) {
822 res = f.invoke_flag(option_position + first_index);
823 } else {
824 for (const auto& vt : vals) {
825 if (res.has_error()) break;
826 res = f.invoke_action(vt.text, vt.position + first_index);
827 }
828 }
829 if (!res.has_error()) f.fire_on_parse();
830 }
831 });
832 return res;
833 }

835 // Fill the pos_idx-th positional field with text. Advances pos_idx/pos_cnt
836 // when the current field reaches its nargs.max.
837 static auto dispatch_positional(T& val, std::size_t& pos_idx,
838 std::size_t& pos_cnt, std::string_view text,
839 std::size_t position) -> ActionResult<void> {
840 ActionResult<void> res = ActionResult<void>::ok();
841 bool invoked = false;
842 std::size_t cur = 0;
843 for_each_field(val, [&](auto& f) -> auto {
844 using F = std::remove_cvref_t<decltype(f)>;
845 if constexpr (std::derived_from<F, PositionalTag>) {
846 if (cur++ != pos_idx || invoked) return;
847 invoked = true;
848 res = f.invoke_action(text, position);
849 if (res.has_value()) {
850 ++pos_cnt;
851 if (F::nargs.max != -1 &&
852 pos_cnt >= static_cast<std::size_t>(F::nargs.max)) {
853 ++pos_idx;
854 pos_cnt = 0;
855 }
856 }
857 }
858 });
859 if (!invoked)
860 return ActionResult<void>::fail(
861 ParseError{.code = ErrorCode::unexpected_argument,
862 .kind = ErrorKind::parse,
863 .position = static_cast<int>(position),
864 .subject = std::string(text)});
865 return res;
866 }

868 auto dispatch_command(T& val, std::string_view name,
869 std::span<const std::string_view> args,
870 std::size_t first_index) -> ActionResult<void> {
871 ActionResult<void> res = ActionResult<void>::ok();
872 bool found = false;
873 for_each_field(val, [&](auto& f) -> auto {
874 using F = std::remove_cvref_t<decltype(f)>;
875 if constexpr (std::derived_from<F, CommandTag>) {
876 if (found || F::command_name() != name) {
877 return;
878 }
879 found = true;
880 f.mark_provided();
881 auto sub_parser = Parser<typename F::argument_type>{};
882 sub_parser.program_name_ =
883 std::format("{} {}", program_name_, F::command_name());
884 auto sub = sub_parser.parse(args, first_index);
885 if (sub.has_error()) {
886 res = ActionResult<void>::fail(std::move(sub.error));
887 } else {
888 static_cast<typename F::argument_type&>(f) = std::move(sub.value);
889 f.mark_provided();
890 }
891 }
892 });
893 if (!found) {
894 return ActionResult<void>::fail(ParseError{
895 .code = ErrorCode::unknown_command,
896 .kind = ErrorKind::parse,
897 .position = static_cast<int>(first_index - 1),
898 .subject = std::string(name),
899 });
900 }
901 return res;
902 }

904 static auto validate_required(T& val) -> ActionResult<void> {
905 ActionResult<void> res = ActionResult<void>::ok();
906 for_each_field(val, [&](auto& f) -> auto {
907 using F = std::remove_cvref_t<decltype(f)>;
908 if constexpr (std::derived_from<F, OptionTag> ||
909 std::derived_from<F, PositionalTag>) {
910 if (res.has_error()) {
911 return;
912 }
913 if (f.presence == Presence::required && !f.provided()) {
914 std::string subject;
915 if constexpr (std::derived_from<F, OptionTag>) {
916 subject = "--" + std::string(F::name.view());
917 } else {
918 subject = "<positional>";
919 }
920 res = ActionResult<void>::fail(ParseError{
921 .code = ErrorCode::missing_required,
922 .kind = ErrorKind::parse,
923 .subject = std::move(subject),
924 });
925 }
926 }
927 });
928 return res;
929 }

931 auto apply_env_fallback(T& val) -> ActionResult<void> {
932 ActionResult<void> res = ActionResult<void>::ok();
933 for_each_field(val, [&](auto& f) -> auto {
934 using F = std::remove_cvref_t<decltype(f)>;
935 if constexpr (std::derived_from<F, OptionTag>) {
936 if (res.has_error() || f.provided() || f.env.empty()) return;
937#ifdef _WIN32
938 char* raw_buf = nullptr;
939 std::size_t raw_len = 0;
940 _dupenv_s(&raw_buf, &raw_len, std::string(f.env).c_str());
941 std::unique_ptr<char, decltype(&free)> raw_guard(raw_buf, &free);
942 const char* raw = raw_buf;
943#else
944 const char* raw = std::getenv(std::string(f.env).c_str());
945#endif
946 if (!raw) return;
947 f.notify_option_seen();
948 if constexpr (F::nargs.min == 0 && F::nargs.max == 0) {
949 res = f.invoke_flag(0);
950 } else {
951 res = f.invoke_action(std::string_view(raw), 0);
952 }
953 if (!res.has_error()) f.fire_on_parse();
954 }
955 });
956 return res;
957 }

960 if (error.code == ErrorCode::help_requested && error.detail.empty()) {
961 error.detail = format_help(ColorMode::never);
962 }
963 return error;
964 }
