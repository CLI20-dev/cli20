412template <class OptionLookup, class CommandLookup>
413inline auto tokenize(std::span<const std::string_view> args,
414 OptionLookup option_nargs, CommandLookup is_command,
415 const TokenizerConfig& cfg = {}) -> TokenizeResult {
416 TokenizeResult result;
417
418 // Returns an encoded prefix tag that can be decoded with token_prefix_view().
419 const auto find_prefix = [&](std::string_view tok) -> std::uint8_t {
420 for (std::uint8_t pi = 0;
421 pi < static_cast<uint8_t>(cfg.option_prefixes().size()); ++pi) {
422 const auto& p = cfg.option_prefixes()[pi];
423 if (tok.size() > p.size() && tok.starts_with(p)) {
424 return long_prefix(pi);
425 }
426 }
427
428 if (tok.size() > cfg.short_option_prefix().size() &&
429 tok.starts_with(cfg.short_option_prefix())) {
430 return short_prefix();
431 }
432
433 return no_prefix();
434 };
435
436 const auto prefix_view = [&](std::uint8_t p) -> std::string_view {
437 if (p == no_prefix()) return {};
438 if (is_short_prefix(p)) return cfg.short_option_prefix();
439 return cfg.option_prefixes()[p & kPrefixIndexMask];
440 };
441
442 const auto push = [&](TokenType type, std::string_view text, std::size_t pos,
443 std::uint8_t prefix = kNoPrefix) -> void {
444 result.tokens.push_back({
445 .text = text,
446 .position = pos,
447 .prefix = prefix,
448 .type = type,
449 });
450 };
451
452 bool post_separator = false;
453 std::size_t i = 0;
454
455 while (i < args.size()) {
456 const std::string_view tok = args[i];
457
458 // ── option parsing terminator ─────────────────────────────────────────
459 if (!post_separator && tok == cfg.end_of_options_separator()) {
460 post_separator = true;
461 ++i;
462 continue;
463 }
464
465 // ── after separator: everything is positional ─────────────────────────
466 if (post_separator) {
467 push(TokenType::positional, tok, i++);
468 continue;
469 }
470
471 // ── command name ──────────────────────────────────────────────────────
472 const auto prefix = find_prefix(tok);
473 const auto prefix_sv = prefix_view(prefix);
474 if (!has_prefix(prefix) && is_command(tok)) {
475 push(TokenType::command, tok, i);
476 break; // args[i+1:] are the sub-command's responsibility
477 }
478
479 // ── option-looking token ──────────────────────────────────────────────
480 if (has_prefix(prefix)) {
481 const auto sep = cfg.inline_value_separator();
482 const auto eq_pos = tok.find(sep);
483 const bool has_inline_val = (eq_pos != std::string_view::npos);
484 const std::string_view opt_sv =
485 has_inline_val ? tok.substr(0, eq_pos) : tok;
486 const auto nargs_opt = option_nargs(opt_sv);
487 if (!nargs_opt) {
488 // Short cluster (-xvf) or attached value (-ofile): only when the token
489 // uses the short prefix and carries more than one character after it.
490 const std::string_view body = opt_sv.substr(prefix_sv.size());
491 if (is_short_prefix(prefix) && !has_inline_val && body.size() > 1) {
492 bool advanced_i = false;
493 const std::string_view char_prefix = tok.substr(0, prefix_sv.size());
494 std::string char_full{char_prefix};
495 char_full.push_back('\0');
496 for (std::size_t k = 0; k < body.size(); ++k) {
497 const std::string_view char_bare =
498 tok.substr(prefix_sv.size() + k, 1);
499 char_full.back() = body[k];
500 const auto char_nargs_opt = option_nargs(char_full);
501 if (!char_nargs_opt) {
502 return result.fail(ErrorCode::unknown_option, i, char_full);
503 }
504 const Nargs nargs = *char_nargs_opt;
505 const bool is_flag = (nargs.min == 0 && nargs.max == 0);
506 push(TokenType::option, char_bare, i, prefix);
507 if (!is_flag) {
508 if (k + 1 < body.size()) {
509 // First value is attached (rest of body).
510 push(TokenType::value, tok.substr(prefix_sv.size() + k + 1), i);
511 ++i;
512 // Consume additional tokens for nargs > 1.
513 int count = 1;
514 while (i < args.size()) {
515 const std::string_view next = args[i];
516 if (next == cfg.end_of_options_separator()) break;
517 if (is_command(next)) break;
518 if (nargs.max != -1 && count >= nargs.max) break;
519 if (has_prefix(find_prefix(next))) {
520 const auto next_sep = next.find(sep);
521 const std::string_view next_opt =
522 next_sep != std::string_view::npos
523 ? next.substr(0, next_sep)
524 : next;
525 if (option_nargs(next_opt).has_value()) break;
526 if (count >= 1 || nargs.min == 0) break;
527 }
528 push(TokenType::value, args[i], i);
529 ++i;
530 ++count;
531 }
532 if (count < nargs.min) {
533 return result.fail(
534 ErrorCode::missing_value, i, char_full,
535 detail::value_count_message(nargs.min, count));
536 }
537 } else {
538 // Consume next token(s) greedily as values
539 ++i;
540 int count = 0;
541 while (i < args.size()) {
542 const std::string_view next = args[i];
543 if (next == cfg.end_of_options_separator()) break;
544 if (is_command(next)) break;
545 if (nargs.max != -1 && count >= nargs.max) break;
546 if (has_prefix(find_prefix(next))) {
547 const auto next_sep = next.find(sep);
548 const std::string_view next_opt =
549 next_sep != std::string_view::npos
550 ? next.substr(0, next_sep)
551 : next;
552 if (option_nargs(next_opt).has_value()) break;
553 if (count >= 1 || nargs.min == 0) break;
554 }
555 push(TokenType::value, args[i], i);
556 ++i;
557 ++count;
558 }
559 if (count < nargs.min) {
560 return result.fail(
561 ErrorCode::missing_value, i, char_full,
562 detail::value_count_message(nargs.min, count));
563 }
564 }
565 advanced_i = true;
566 break;
567 }
568 }
569 if (!advanced_i) ++i;
570 continue;
571 }
572 return result.fail(ErrorCode::unknown_option, i, std::string(opt_sv));
573 }
574
575 const Nargs& nargs = *nargs_opt;
576 const bool is_flag = (nargs.min == 0 && nargs.max == 0);
577
578 // Store bare name (without prefix) in text; matched_prefix holds the
579 // prefix.
580
581 push(TokenType::option, opt_sv.substr(prefix_sv.size()), i, prefix);
582
583 if (has_inline_val) {
584 if (is_flag) {
585 return result.fail(ErrorCode::invalid_value, i, std::string(opt_sv),
586 "flag does not accept a value");
587 }
588 if (nargs.min > 1) {
589 return result.fail(ErrorCode::missing_value, i, std::string(opt_sv),
590 detail::inline_value_count_message(nargs.min));
591 }
592 push(TokenType::value, tok.substr(eq_pos + 1), i);
593 ++i;
594 continue;
595 }
596
597 // prefix-opt form: advance, then greedily consume value tokens
598 ++i;
599
600 if (is_flag) continue;
601
602 int count = 0;
603 while (i < args.size()) {
604 const std::string_view next = args[i];
605
606 if (next == cfg.end_of_options_separator()) break;
607 if (is_command(next)) break;
608 if (nargs.max != -1 && count >= nargs.max) break;
609
610 if (has_prefix(find_prefix(next))) {
611 const auto next_sep = next.find(sep);
612 const std::string_view next_opt = next_sep != std::string_view::npos
613 ? next.substr(0, next_sep)
614 : next;
615 if (option_nargs(next_opt).has_value()) break;
616
617 // Unknown option-looking: consume only as first value when min > 0
618 if (count >= 1 || nargs.min == 0) break;
619 }
620
621 push(TokenType::value, args[i], i);
622 ++i;
623 ++count;
624 }
625
626 if (count < nargs.min) {
627 return result.fail(ErrorCode::missing_value, i, std::string(opt_sv),
628 detail::value_count_message(nargs.min, count));
629 }
630 continue;
631 }
632
633 // ── bare token → positional ───────────────────────────────────────────
634 push(TokenType::positional, tok, i++);
635 }
636
637 return result;
638}

880 auto parse_body(ParseResult<T>& result, std::span<const std::string_view> args,
881 std::size_t first_index) -> void {
882 auto lookup = get_option_spec_map(result.value);
883
884 auto tokenized = tokenize(
885 args.subspan(std::min(first_index, args.size())),
886 [&](std::string_view opt) -> auto { return lookup.option_nargs(opt); },
887 [&](std::string_view name) -> auto { return lookup.is_command(name); },
888 cfg_);
889 if (tokenized.has_error()) {
890 if (tokenized.error.has_position()) {
891 tokenized.error.position += static_cast<int>(first_index);
892 }
893 result.error = std::move(tokenized.error);
894 return;
895 }
896
897 const auto& tokens = tokenized.tokens;
898 std::size_t pos_idx = 0, pos_cnt = 0;
899
900 for (std::size_t i = 0; i < tokens.size();) {
901 const Token& tok = tokens[i];
902
903 if (tok.type == TokenType::option) {
904 std::size_t j = i + 1;
905 while (j < tokens.size() && tokens[j].type == TokenType::value) ++j;
906 auto err =
907 dispatch_option(result.value, token_prefix_view(tok, cfg_), tok.text,
908 std::span(tokens).subspan(i + 1, j - i - 1),
909 first_index, tok.position);
910 if (err.has_error()) {
911 result.error = finalize_special_error(std::move(err.error));
912 return;
913 }
914 i = j;
915
916 } else if (tok.type == TokenType::positional) {
917 auto err = dispatch_positional(result.value, pos_idx, pos_cnt, tok.text,
918 tok.position + first_index);
919 if (err.has_error()) {
920 result.error = finalize_special_error(std::move(err.error));
921 return;
922 }
923 ++i;
924
925 } else if (tok.type == TokenType::command) {
926 auto err = dispatch_command(result.value, tok.text, args,
927 tok.position + first_index + 1);
928 if (err.has_error()) {
929 result.error = finalize_special_error(std::move(err.error));
930 return;
931 }
932 break;
933
934 } else {
935 ++i;
936 }
937 }
938
939 // Fire on_parse callbacks for positionals once, after all tokens are
940 // consumed.
941 for_each_field(result.value, [](auto& f) -> auto {
942 using F = std::remove_cvref_t<decltype(f)>;
943 if constexpr (std::derived_from<F, PositionalTag>) {
944 if (f.provided()) f.on_parse();
945 }
946 });
947
948 if (auto err = apply_env_fallback(result.value); err.has_error()) {
949 result.error = std::move(err.error);
950 return;
951 }
952
953 if (auto err = validate_required(result.value); err.has_error()) {
954 result.error = std::move(err.error);
955 return;
956 }
957
958 if (auto err = validate_relations(result.value); err.has_error()) {

1087 [[nodiscard]] auto option_nargs(std::string_view opt) const
1088 -> std::optional<Nargs> {
1089 for (std::size_t i = 0; i < option_count; ++i) {
1090 const auto& entry = options[i];
1091 if (!opt.starts_with(entry.prefix)) {
1092 continue;
1093 }
1094 const auto rest = opt.substr(entry.prefix.size());
1095 if (entry.is_short) {
1096 if (rest.size() == 1 && rest.front() == entry.short_name) {
1097 return entry.nargs;
1098 }
1099 } else if (rest == entry.name) {
1100 return entry.nargs;
1101 }
1102 }
1103 return std::nullopt;
1104 }

1107 for (std::size_t i = 0; i < command_count; ++i) {
1108 if (commands[i].name == name) return true;

1110 return false;

1133 // Find the option field matching (prefix, bare), notify it, and invoke its
1134 // action on each value token. notify_option_seen() is called once;
1135 // invoke_action() is called once for flags / missing optional entries, or
1136 // once per value token otherwise.
1137 auto dispatch_option(T& val, std::string_view prefix, std::string_view bare,
1138 std::span<const Token> vals, std::size_t first_index,
1139 std::size_t option_position) -> ActionResult<void> {
1140 ActionResult<void> res = ActionResult<void>::ok();
1141 bool found = false;
1142 for_each_field(val, [&](auto& f) -> auto {
1143 using F = std::remove_cvref_t<decltype(f)>;
1144 if constexpr (std::derived_from<F, OptionTag>) {
1145 if (found) {
1146 return;
1147 }
1148 bool match = false;
1149 for (const auto& p : cfg_.option_prefixes()) {
1150 if (p == prefix && F::name.view() == bare) {
1151 match = true;
1152 break;
1153 }
1154 }
1155 if (!match && F::short_name != '\0' &&
1156 cfg_.short_option_prefix() == prefix && bare.size() == 1 &&
1157 bare[0] == F::short_name) {
1158 match = true;
1159 }
1160 if (!match) {
1161 return;
1162 }
1163 found = true;
1164 f.notify_option_seen();
1165 if constexpr (F::entry_is_optional) {
1166 res = f.invoke_action(std::nullopt, option_position + first_index,
1167 vals.size(), 0);
1168 }
1169 const std::size_t nargs = vals.size();
1170 for (std::size_t vi = 0; vi < vals.size(); ++vi) {
1171 if (res.has_error()) {
1172 break;
1173 }
1174 res = f.invoke_action(vals[vi].text, vals[vi].position + first_index,
1175 nargs, vi);
1176 }
1177 if (!res.has_error()) f.on_parse();
1178 }
1179 });
1180 return res;
1181 }

1183 // Fill the pos_idx-th positional field with text. Advances pos_idx/pos_cnt
1184 // when the current field reaches its nargs.max.
1185 static auto dispatch_positional(T& val, std::size_t& pos_idx,
1186 std::size_t& pos_cnt, std::string_view text,
1187 std::size_t position) -> ActionResult<void> {
1188 ActionResult<void> res = ActionResult<void>::ok();
1189 bool invoked = false;
1190 std::size_t cur = 0;
1191 for_each_field(val, [&](auto& f) -> auto {
1192 using F = std::remove_cvref_t<decltype(f)>;
1193 if constexpr (std::derived_from<F, PositionalTag>) {
1194 if (cur++ != pos_idx || invoked) return;
1195 invoked = true;
1196 res = f.invoke_action(text, position);
1197 if (res.has_value()) {
1198 ++pos_cnt;
1199 if (F::nargs.max != -1 &&
1200 pos_cnt >= static_cast<std::size_t>(F::nargs.max)) {
1201 ++pos_idx;
1202 pos_cnt = 0;
1203 }
1204 }
1205 }
1206 });
1207 if (!invoked)
1208 return ActionResult<void>::fail(
1209 ParseError{.code = ErrorCode::unexpected_argument,
1210 .kind = ErrorKind::parse,
1211 .position = static_cast<int>(position),
1212 .subject = std::string(text)});
1213 return res;
1214 }

1318 static auto relation_operand_present(T& val, std::string_view name) -> bool {
1319 bool found_field = false;
1320 bool present = false;
1321 for_each_field(val, [&](auto& f) -> auto {
1322 using F = std::remove_cvref_t<decltype(f)>;
1323 if constexpr (std::derived_from<F, OptionTag> ||
1324 std::derived_from<F, CommandTag>) {

1326 found_field = true;
1327 present = f.provided();
1328 }
1329 }
1330 });

1337 [&](auto... rels) -> auto {

1346 group_present = group_present &&

1360 static auto validate_relations(T& val) -> ActionResult<void> {
1361 if constexpr (!detail::HasRelations<T>) {
1362 return ActionResult<void>::ok();
1363 } else {
1364 ActionResult<void> res = ActionResult<void>::ok();
1365 std::apply(
1366 [&](auto... rels) -> auto {
1367 (
1368 [&]() -> auto {
1369 if (res.has_error()) return;
1370 using R = std::remove_cvref_t<decltype(rels)>;
1371 if constexpr (std::same_as<R, GroupRelation>) {
1372 std::size_t count = 0;
1373 for (auto member : rels.members) {
1374 if (relation_operand_present(val, member)) ++count;
1375 }
1376 if (count != 0 && count != rels.members.size) {
1377 res = ActionResult<void>::fail(ParseError{
1378 .code = ErrorCode::dependency_missing,
1379 .kind = ErrorKind::validation,
1380 .subject = std::string(rels.name),
1381 .detail = "arguments must be provided together: " +
1382 relation_name_list(rels.members),
1383 });
1384 }
1385 } else if constexpr (std::same_as<R, ConflictsRelation>) {
1386 if (relation_operand_present(val, rels.left) &&
1387 relation_operand_present(val, rels.right)) {
1388 res = ActionResult<void>::fail(ParseError{
1389 .code = ErrorCode::mutually_exclusive,
1390 .kind = ErrorKind::validation,
1391 .subject = relation_operand_label(rels.left) + ", " +
1392 relation_operand_label(rels.right),
1393 .detail =
1394 "these arguments cannot be provided together",
1395 });
1396 }
1397 } else if constexpr (std::same_as<R, DependsOnRelation>) {
1398 if (relation_operand_present(val, rels.source) &&
1399 !relation_operand_present(val, rels.target)) {
1400 res = ActionResult<void>::fail(ParseError{
1401 .code = ErrorCode::dependency_missing,
1402 .kind = ErrorKind::validation,
1403 .subject = relation_operand_label(rels.source),
1404 .detail =
1405 "requires " + relation_operand_label(rels.target),
1406 });
1407 }
1408 }
1409 }(),
1410 ...);
1411 },
1412 T::relations.items);
1413 return res;
1414 }
